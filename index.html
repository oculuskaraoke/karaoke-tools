🎤 Karaoke Syllabifier

Paste your lyrics below to automatically split syllables with slashes.

packages = ["pyphen"] import pyphen from pyscript import document def process_lyrics(event): # left=1 and right=1 force the dictionary to split single letters like "a/round" dic = pyphen.Pyphen(lang='en_US', left=1, right=1) input_text = document.getElementById('input_box').value processed_lines = [] for line in input_text.split('\n'): words = line.split() syllabified_words = [] for word in words: # Split syllables and replace hyphen with slash syllabified_word = dic.inserted(word).replace('-', '/') syllabified_words.append(syllabified_word) processed_lines.append(" ".join(syllabified_words)) document.getElementById('output_box').value = "\n".join(processed_lines)
