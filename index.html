🎤 Karaoke Syllabifier

Paste your lyrics below to automatically split syllables with slashes.

packages = ["pyphen"] import pyphen from pyscript import document def process_lyrics(event): # Load the dictionary dic = pyphen.Pyphen(lang='en_US') # Get text from the input box input_text = document.getElementById('input_box').value processed_lines = [] for line in input_text.split('\n'): words = line.split() syllabified_words = [] for word in words: # Split syllables and replace hyphen with slash syllabified_word = dic.inserted(word).replace('-', '/') syllabified_words.append(syllabified_word) processed_lines.append(" ".join(syllabified_words)) # Push the result to the output box document.getElementById('output_box').value = "\n".join(processed_lines)
